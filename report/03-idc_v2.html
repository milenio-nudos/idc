Índice general

# 2. Convertir a formato largo
comunas_long <- comunas_filtradas %>%
  pivot_longer(
    cols = c(ranking_idc_social_2024, ranking_2025),
    names_to = "anio",
    values_to = "ranking"
  ) %>%
  mutate(anio = ifelse(anio == "ranking_idc_social_2024", "2024", "2025"))

# 3. Ordenar comunas por ranking 2025 para visualización clara
orden_comunas <- comunas_filtradas %>%
  arrange(ranking_2025) %>%
  pull(m_name)

# 4. Crear gráfico tipo lollipop horizontal

  scale_x_reverse() +  # Ranking más alto a la izquierda

    color = "Año"
  ) +
  theme_minimal(base_size = 10) +

datos_completos <- comparaciones_educacion %>%

# Calcular correlación
r_valor <- cor(datos_completos$m_idc_2024, datos_completos$m_idc_2025, use = "complete.obs") %>% round(2)

# Graficar
ggplot(datos_completos, aes(x = m_idc_2024, y = m_idc_2025)) +
  geom_point(color = "#6495ED", alpha = 0.7) +
  geom_smooth(method = "lm", se = FALSE, color = "darkred", linetype = "dashed") +
  annotate("text", x = Inf, y = -Inf, hjust = 1.1, vjust = -1,
           label = paste0("r = ", r_valor), size = 5, color = "black") +
  labs(
    x = "índice educacional 2024 (Todos los estamentos)",
    y = "Índice educacional 2025",

    subtitle = paste0("n = ", nrow(datos_completos), " comunas"),

  ) +
